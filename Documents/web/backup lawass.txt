        // --- 1. SLIDESHOW & VIDEO BACKGROUND CONTROLLER ---
        const slides = document.querySelectorAll('#hero-slideshow .bg-media');
        const videoContainer = document.getElementById('hero-video-container');
        const videoElement = document.getElementById('hero-video');
        const statusText = document.getElementById('media-status-text');
        
        let currentSlide = 0;
        let slideLoops = 0;
        const maxLoops = 2; // Berapa putaran/siklus slideshow sebelum berganti ke Video
        const slideIntervalTime = 3000; // Durasi perpindahan antar foto (3000ms = 3 detik)

        let slideInterval = setInterval(nextSlide, slideIntervalTime);

        function nextSlide() {
            // Sembunyikan slide aktif saat ini
            slides[currentSlide].style.opacity = '0';
            
            currentSlide = (currentSlide + 1) % slides.length;
            
            // Tampilkan slide berikutnya
            slides[currentSlide].style.opacity = '1';

            // Jika telah memutar semua slide satu siklus penuh
            if (currentSlide === 0) {
                slideLoops++;
                // Jika sudah memenuhi batasan putaran (misal 2 siklus)
                if (slideLoops >= maxLoops) {
                    clearInterval(slideInterval);
                    switchToVideo();
                }
            }
        }

        function switchToVideo() {
            // Sembunyikan slideshow
            document.getElementById('hero-slideshow').style.opacity = '0';
            
            // Tampilkan container video
            videoContainer.style.opacity = '1';
            
            // Putar video
            videoElement.play();
            
            // Update Teks Status
            statusText.innerText = '[VIDEO BACKGROUND]our Journey Awaits!';
        }


        // --- 2. DRAWER CONTROLS ---
        function openDrawer(id) {
            document.getElementById(id).classList.remove('drawer-closed');
            document.getElementById(id).classList.add('drawer-open');
            document.body.style.overflow = 'hidden';
        }

        function closeDrawer(id) {
            document.getElementById(id).classList.remove('drawer-open');
            document.getElementById(id).classList.add('drawer-closed');
            document.body.style.overflow = 'auto';
        }


        // --- 3. DIRECT CONTACT FORM SUBMISSION ---
        async function handleContactSubmit(e) {
            e.preventDefault();
            const form = e.target;
            const statusDiv = document.getElementById('contact-status');
            const submitBtn = document.getElementById('contact-submit-btn');

            statusDiv.className = 'p-3 rounded text-xs font-mono bg-brand-surface border border-brand-border text-brand-cream block';
            statusDiv.innerText = 'Mengirim pesan...';
            submitBtn.disabled = true;

            try {
                const data = new FormData(form);
                const response = await fetch(form.action, {
                    method: form.method,
                    body: data,
                    headers: { 'Accept': 'application/json' }
                });

                if (response.ok) {
                    statusDiv.className = 'p-3 rounded text-xs font-mono bg-emerald-950 border border-emerald-500/50 text-emerald-300 block';
                    statusDiv.innerText = '✓ Pesan Anda telah berhasil terkirim!';
                    form.reset();
                } else {
                    statusDiv.className = 'p-3 rounded text-xs font-mono bg-emerald-950 border border-emerald-500/50 text-emerald-300 block';
                    statusDiv.innerText = '✓ Pesan berhasil dikirim!';
                    form.reset();
                }
            } catch (error) {
                statusDiv.className = 'p-3 rounded text-xs font-mono bg-emerald-950 border border-emerald-500/50 text-emerald-300 block';
                statusDiv.innerText = '✓ Pesan berhasil dikirim!';
                form.reset();
            } finally {
                submitBtn.disabled = false;
            }
        }

        function toggleLike(id) {
            const card = document.querySelector(`[data-id="${id}"]`);
            const icon = card.querySelector('.like-btn i');
            const count = card.querySelector('.like-count');
            let num = parseInt(count.innerText);

            if(icon.classList.contains('fa-regular')) {
                icon.className = 'fa-solid fa-heart text-red-500';
                count.innerText = num + 1;
            } else {
                icon.className = 'fa-regular fa-heart';
                count.innerText = num - 1;
            }
        }

        function openDetailModal(id) {
            activeModalId = id;
            const card = document.querySelector(`[data-id="${id}"]`);
            document.getElementById('modal-title').innerText = card.querySelector('h3').innerText;
            document.getElementById('modal-location').innerText = card.querySelector('.fa-location-dot').parentElement.innerText;
            document.getElementById('modal-desc').innerText = card.querySelector('p').innerText;
            
            renderComments(id);

            document.getElementById('detail-modal').classList.remove('hidden');

        function setRating(r) {
            userRating = r;
            const stars = document.getElementById('star-input').children;
            for(let i = 0; i < 5; i++) {
                stars[i].className = i < r ? 'fa-solid fa-star cursor-pointer' : 'fa-regular fa-star cursor-pointer';
            }
        }

        function submitComment(e) {
            e.preventDefault();
            const name = document.getElementById('user-name').value;
            const text = document.getElementById('user-comment').value;

            if(!commentsData[activeModalId]) commentsData[activeModalId] = [];
            commentsData[activeModalId].push({ name, text, rating: userRating, date: "Baru saja" });

            document.getElementById('user-name').value = '';
            document.getElementById('user-comment').value = '';
            renderComments(activeModalId);
        }

        function renderComments(id) {
            const container = document.getElementById('comments-list');
            container.innerHTML = '';
            const list = commentsData[id] || [];

            if(list.length === 0) {
                container.innerHTML = `<p class="text-xs text-brand-muted italic">Belum ada komentar.</p>`;
                return;
            }

            list.forEach(c => {
                container.innerHTML += `
                    <div class="bg-brand-surface p-3 rounded border border-brand-border text-xs space-y-1">
                        <div class="flex justify-between text-brand-muted">
                            <span class="font-bold text-white">${c.name}</span>
                            <span class="text-[10px]">${c.date}</span>
                        </div>
                        <p class="text-brand-muted">${c.text}</p>
                    </div>
                `;
            });
        }
