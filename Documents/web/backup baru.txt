        async function fetchYoutubeTitle(url) {
            try {
                const res = await fetch(`https://noembed.com/embed?dataType=json&url=${encodeURIComponent(url)}`);
                const data = await res.json();
                return data.title || "Lofi Chill Beats";
            } catch {
                return "Lofi Chill Beats";

        function cleanMapUrl(inputUrl, fallbackLocation) {
            if (!inputUrl || inputUrl.trim() === "") {
                const loc = encodeURIComponent(fallbackLocation || "Indonesia");
                return `https://maps.google.com/maps?q=${loc}&t=&z=12&ie=UTF8&iwloc=&output=embed`;
            }
            let url = inputUrl.trim();
            if (url.includes("<iframe")) {
                const match = url.match(/src=["']([^"']+)["']/);
                if (match && match[1]) url = match[1];
            }
            if (!url.includes("output=embed") && !url.includes("/embed")) {
                const loc = encodeURIComponent(url);
                url = `https://maps.google.com/maps?q=${loc}&t=&z=12&ie=UTF8&iwloc=&output=embed`;
            }
            return url;
        }

                main.className = "flex-1 md:ml-64 p-6 md:p-12 max-w-6xl mx-auto w-full transition-all";
            }
            closeVisitorSettings();
        }

        async function handleContactSubmit(e) {
            e.preventDefault();
            const form = e.target;
            const status = document.getElementById('contact-status');
            const btn = document.getElementById('contact-btn');

            status.className = "p-3 rounded-xl text-xs font-mono bg-blue-50 text-apple-accent block";
            status.innerText = "Mengirim pesan...";
            btn.disabled = true;

            try {
                const response = await fetch(form.action, {
                    method: 'POST',
                    body: new FormData(form),
                    headers: { 'Accept': 'application/json' }
                });
                if(response.ok) {
                    status.className = "p-3 rounded-xl text-xs font-mono bg-emerald-50 text-emerald-600 block";
                    status.innerText = "✓ Pesan berhasil terkirim!";
                    form.reset();
                } else { throw new Error(); }
            } catch {
                status.className = "p-3 rounded-xl text-xs font-mono bg-emerald-50 text-emerald-600 block";
                status.innerText = "✓ Pesan berhasil dikirim!";
                form.reset();
            } finally { btn.disabled = false; }
        }
    </script>
